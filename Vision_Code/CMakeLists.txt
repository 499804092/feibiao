## ============================================================================
## 【本任务改动说明】
##   1) cmake_minimum_required 由 3.25 放宽到 3.20
##      Ubuntu 22.04 自带 CMake 只有 3.22.1，原来直接报
##      "CMake 3.25 or higher is required"，配置都过不去。
##      本工程用到的特性在 3.20 就有，放宽没有副作用。
##   2) 录像模块（Capture / mcap）改成可选，默认关闭 → ENABLE_CAPTURE=OFF
##      原因：它依赖 FlatBuffers 的 CMake 包和硬编码路径的 lz4/zstd：
##          /usr/lib/x86_64-linux-gnu/liblz4.so.1.9.2
##          /usr/lib/x86_64-linux-gnu/libzstd.so.1.4.4
##      而 Ubuntu 22.04 上是 liblz4.so.1.9.3 / libzstd.so.1.4.8，
##      FlatBuffers 也没装 CMake 配置。为了"相机能出画面、检测能跑"，
##      这些录像依赖不必拖住整个工程。
##   3) 如果你的目标是**录像/回放**（写 mcap 给 Foxglove 看），恢复方式：
##        sudo apt install libflatbuffers-dev flatbuffers-compiler liblz4-dev libzstd-dev
##        cd build && cmake .. -DENABLE_CAPTURE=ON && make -j$(nproc)
##      并把 src/capture/CMakeLists.txt 里那两行写死的 .so 版本号换成实际路径：
##        ls /usr/lib/x86_64-linux-gnu/liblz4.so*  libzstd.so*
## ============================================================================
cmake_minimum_required(VERSION 3.20)
project(Dart)

option(ENABLE_CAPTURE "编译录像/回放模块（需要 FlatBuffers + lz4 + zstd）" OFF)

# ---------------------------------------------------------------------------
# 【新增】DART_NO_FLATBUFFERS —— 给 ARM 开发板用的开关
#   为什么需要：开发板（Radxa ZERO 3W 等）上 flatbuffers 的开发包不一定装得上，
#   或者版本和 config/autogenerated_flatbuffers 里 1.12 生成的头对不上。
#   而默认构建（ENABLE_CAPTURE=OFF）其实**根本用不到 flatbuffers** ——
#   整个工程只用 foxglove::Time 的 sec/nsec，见 config/autogenerated_flatbuffers/Time_generated.h。
#
#   打开这个开关（cmake .. -DDART_NO_FLATBUFFERS=ON）：
#     全工程都不再 include flatbuffers，Time 用头文件里的极简替身。
#   开发板上装不上 flatbuffers 时用它，能少装一大堆东西。
# ---------------------------------------------------------------------------
option(DART_NO_FLATBUFFERS "不依赖 flatbuffers，用极简的 foxglove::Time 替身（开发板用）" OFF)
if(DART_NO_FLATBUFFERS)
  # add_compile_definitions 会影响它【之后】定义的所有目标（utility/detect/track/...），
  # 所以必须放在下面那些 add_subdirectory 之前。
  add_compile_definitions(DART_NO_FLATBUFFERS=1)
  message(STATUS "DART_NO_FLATBUFFERS=ON —— 用极简 Time 替身，全工程不依赖 flatbuffers")
endif()

aux_source_directory(${CMAKE_CURRENT_SOURCE_DIR} ${PROJECT_NAME}_src)

add_executable(${PROJECT_NAME} ${${PROJECT_NAME}_src})

add_subdirectory(src/utility)

if(ENABLE_CAPTURE)
  message(STATUS "ENABLE_CAPTURE=ON —— 编译录像模块（需要 FlatBuffers/lz4/zstd）")
  add_subdirectory(src/capture)
else()
  message(STATUS "ENABLE_CAPTURE=OFF —— 跳过录像模块（相机/检测/跟踪/串口不受影响）")
endif()

add_subdirectory(src/sensors)
add_subdirectory(src/detect)
add_subdirectory(src/track)
# 【改动】原来这里是 add_subdirectory(src/mouse)。
#   mouse 是原版比赛代码留下的"用鼠标手动画裁剪框"工具，它只从队列里取原图、
#   把识别结果丢掉，对识别和跟踪没有任何贡献，只会弹一个让人误解的窗口。
#   已经整体移到 beifen/已弃用-mouse调试框/ ，换成真正显示识别结果的 view 模块。
# 【新增】web 模块：MJPEG over HTTP 推流，让没有桌面的开发板也能在电脑浏览器上看画面
#   ★必须放在 view 之前★：view 会 target_link_libraries(... web)，
#   如果 web 目标还没定义，CMake 会把 "web" 当成一个普通库名去找，直接报找不到。
add_subdirectory(src/web)
add_subdirectory(src/view)


target_link_libraries(${PROJECT_NAME} PRIVATE utility)
target_include_directories(${PROJECT_NAME} PRIVATE src/utility/inc)

if(ENABLE_CAPTURE)
  target_compile_definitions(${PROJECT_NAME} PRIVATE ENABLE_CAPTURE=1)
  target_link_libraries(${PROJECT_NAME} PRIVATE capture)
  target_include_directories(${PROJECT_NAME} PRIVATE src/capture/inc)
endif()

target_include_directories(${PROJECT_NAME} PRIVATE src/threads/inc)

target_link_libraries(${PROJECT_NAME} PRIVATE sensors)
target_include_directories(${PROJECT_NAME} PRIVATE src/sensors/inc)

target_link_libraries(${PROJECT_NAME} PRIVATE detect)
target_include_directories(${PROJECT_NAME} PRIVATE src/detect/inc)

target_link_libraries(${PROJECT_NAME} PRIVATE track)
target_include_directories(${PROJECT_NAME} PRIVATE src/track/inc)

target_link_libraries(${PROJECT_NAME} PRIVATE view)
target_include_directories(${PROJECT_NAME} PRIVATE src/view/inc)

target_link_libraries(${PROJECT_NAME} PRIVATE web)
target_include_directories(${PROJECT_NAME} PRIVATE src/web/inc)


target_link_libraries(${PROJECT_NAME} PRIVATE -lpthread)
